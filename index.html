// 增加多个可能的特征值变量 let charWrite, charNotify, allWritableChars = []; async function connect() { try { log("正在搜索 LBS 硬件..."); const device = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: ['0000ff00-0000-1000-8000-00805f9b34fb'] }); const server = await device.gatt.connect(); const service = await server.getPrimaryService('0000ff00-0000-1000-8000-00805f9b34fb'); // 获取该服务下所有的特征值 const chars = await service.getCharacteristics(); allWritableChars = []; for (let c of chars) { log(`发现特征值: ${c.uuid.slice(4,8)} | 属性: ${Object.keys(c.properties).filter(p => c.properties[p]).join(', ')}`); // 记录所有可以写入的通道 if (c.properties.write || c.properties.writeWithoutResponse) { allWritableChars.push(c); } // 默认订阅 FF01 作为数据接收端 if (c.uuid.includes('ff01')) { charNotify = c; await charNotify.startNotifications(); charNotify.addEventListener('characteristicvaluechanged', (e) => { let hex = Array.from(new Uint8Array(e.target.value.buffer)).map(b => b.toString(16).padStart(2, '0')).join(' '); log("🔥 收到反馈: " + hex); }); } } log(`✅ 连接成功！共找到 ${allWritableChars.length} 个可写通道。`); document.getElementById('devStatus').innerText = "已连接: " + device.name; } catch (err) { log("❌ 连接失败: " + err); } } async function autoTry() { if (allWritableChars.length === 0) return log("请先连接并确认有可写通道"); // 扩充后的 LBS 指令库 const testCmds = ['01', 'AA550101', 'AA550102', '55AA0101', '06010101', '00000001']; log("🚀 开始全通道轰炸激活..."); for (let char of allWritableChars) { log(`--- 正在测试通道: ${char.uuid.slice(4,8)} ---`); for (let cmd of testCmds) { try { let bytes = new Uint8Array(cmd.match(/.{1,2}/g).map(byte => parseInt(byte, 16))); // 强制使用 WNR 模式以避免 出错2 await char.writeValueWithoutResponse(bytes); log(`已向 ${char.uuid.slice(4,8)} 发送 ${cmd}`); await new Promise(r => setTimeout(r, 800)); // 稍作停顿观察反馈 } catch (e) { log(`通道 ${char.uuid.slice(4,8)} 写入失败: ${e.message}`); } } } log("🏁 全通道测试结束。"); }
